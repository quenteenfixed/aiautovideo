1. 项目概述

1.1 项目目标

构建一套自动化视频生产流水线：输入手动编排的 script.json 分镜脚本，输出高质量的科普短视频 MP4 文件。支持批量生产，每天产出多个视频且画面风格不重复。

1.2 核心需求

6.1 TypeScript 类型定义

interface ScriptData {
  video_id: string;
  title: string;
  resolution: { width: number; height: number };
  fps: number;                          // 30 或 60
  template: TemplateType;                  // 视觉模板名称

  global_style: {
    font_family: string;
    primary_color: string;
    accent_color: string;
    bg_color: string;
    text_color: string;
  };

  voice: {
    provider: 'edge-tts' | 'cosyvoice';
    voice_name: string;              // 如 zh-CN-YunxiNeural
    rate: string;                    // 如 "+10%" / "-20%"
    pitch: string;                   // 如 "0Hz"
    volume: string;                   // 如 "0" (默认)
  };

  bgm: {
    file: string;                      // assets/bgm/ 下的文件名
    volume: number;                    // 0.0 ~ 1.0
    fade_in: number;                  // 秒
    fade_out: number;                 // 秒
  };

  subtitle_style: {
    font_size: number;
    font_family: string;
    color: string;                     // 已读文字颜色
    highlight_color: string;          // 当前词高亮颜色
    stroke_color: string;
    stroke_width: number;
    animation: 'bounce' | 'slide' | 'fade' | 'pop';
    position: 'bottom' | 'center' | 'top';
    offset_y: number;                 // px 距离边缘
    max_width: number;                 // px 字幕最大宽度
  };

  scenes: Scene[];
  outro: Outro;
}

interface Scene {
  scene_id: number;
  narration: string;                    // 旁白文本（TTS 输入）
  duration: number;                     // 预估秒数（TTS 后更新为实际值）

  visual: SceneVisual;

  transition_in: TransitionType;
  transition_out: TransitionType;
}

type SceneVisual =
  | { type: 'text_card'; title: string; subtitle?: string;
      bg_image?: string; animation: string;
      components?: OverlayComponent[] }
  | { type: 'chart'; chart_type: ChartType;
      data: Record<string, any>; animation: string;
      components?: OverlayComponent[] }
  | { type: 'animation'; animation_source: string;
      overlay_components?: OverlayComponent[] }
  | { type: 'image'; image_source: string;
      effect: 'ken_burns' | 'parallax' | 'static';
      overlay_components?: OverlayComponent[] }
  | { type: 'mixed'; elements: MixedElement[] };

interface OverlayComponent {
  type: 'label' | 'arrow' | 'highlight' | 'badge';
  text?: string;
  position: string;                    // 如 "top_center" / {x,y}
  color?: string;
  from?: string; to?: string;       // for arrow
}

type TransitionType = 'fade' | 'slide_left' | 'slide_right'
  | 'slide_up' | 'zoom_in' | 'zoom_out' | 'bounce' | 'wipe' | 'none';

type ChartType = 'bar' | 'line' | pie | 'scatter' | 'radar' | 'spectrum' | 'custom';

type TemplateType = 'data_visual_style' | 'illustration_style'
  | 'cinematic_style' | 'minimal_style' | 'infographic_style';

interface Outro {
  duration: number;
  narration: string;
  visual: { type: 'cta_card'; text: string; animation: string };
  transition_in: TransitionType;
}

7.1 TTS 语音合成模块

功能职责

接收每个场景的旁白文本和语音配置，调用 edge-tts 生成音频，并提取词级时间戳用于字幕同步。

技术方案

项目方案
TTS 引擎Python edge-tts（WebSocket 调用微软 Edge 在线 TTS）
调用方式Node.js 通过 child_process.exec 调用 Python 脚本
词级时间戳edge-tts 的 WordBoundary 事件（offset/duration 单位：100ns）
音色数量500+ 音色（中文 20+ 种）
语速控制--rate "+10%" 参数，范围 -50% ~ +100%
输出格式MP3（24kHz / 48kbps / mono）

Python TTS 脚本 (scripts/tts.py)

import asyncio
import edge_tts
import json
import sys

async def synthesize(text, voice, rate, pitch, volume, output_path):
    communicate = edge_tts.Communicate(text, voice, rate=rate, pitch=pitch, volume=volume)

    audio_chunks = []
    word_boundaries = []

    async for chunk in communicate.stream():
        if chunk["type"] == "audio":
            audio_chunks.append(chunk["data"])
        elif chunk["type"] == "WordBoundary":
            # offset/duration 单位为 100ns，转换为 ms
            word_boundaries.append({
                "text": chunk["text"],
                "offset_ms": chunk["offset"] / 10000,
                "duration_ms": chunk["duration"] / 10000
            })

    # 写入音频文件
    with open(output_path, "wb") as f:
        f.write(b"".join(audio_chunks))

    # 输出 JSON 结果到 stdout
    result = {
        "audio_file": output_path,
        "word_boundaries": word_boundaries,
        "word_count": len(word_boundaries)
    }
    print(json.dumps(result, ensure_ascii=False))

if __name__ == "__main__":
    # 参数: text|voice|rate|pitch|volume|output_path
    args = sys.argv[1:]
    asyncio.run(synthesize(*args))

TypeScript 调用层 (scripts/tts.ts)

import { exec } from 'child_process';
import { promisify } from 'util';
import { getAudioDuration } from './helpers';

const execAsync = promisify(exec);

interface TTSResult {
  audio_file: string;
  audio_duration: number;
  word_boundaries: WordBoundary[];
}

async function synthesizeScene(
  sceneId: number,
  narration: string,
  voice: VoiceConfig,
  outputDir: string
): Promise<TTSResult> {
  const outputFile = `${outputDir}/scene_${sceneId}.mp3`;
  const cmd = `python3 scripts/tts.py "${narration}" `
    + `${voice.voice_name} ${voice.rate} ${voice.pitch} ${voice.volume} ${outputFile}`;

  const { stdout } = await execAsync(cmd);
  const result = JSON.parse(stdout);

  const audio_duration = await getAudioDuration(outputFile);

  return { ...result, audio_duration };
}

export async function generateAllAudio(
  script: ScriptData,
  outputDir: string
): Promise<Map<number, TTSResult>> {
  const results = new Map();

  // 并行合成所有场景音频（edge-tts 支持并发）
  const tasks = script.scenes.map(scene =>
    synthesizeScene(scene.scene_id, scene.narration, script.voice, outputDir)
      .then(result => results.set(scene.scene_id, result))
  );
  await Promise.all(tasks);

  return results;
}

7.2 Remotion 渲染引擎

核心原理

Remotion 将 React 组件渲染为每一帧的画面，通过 Chrome Headless 截图，再用 FFmpeg 编码为视频。每一帧的渲染由 useCurrentFrame() 驱动，是纯函数式的：相同输入永远产生相同输出。

入口文件 (src/index.ts)

根组合 (src/Root.tsx)

    id="MainVideo"
    component={VideoComposition}
    durationInFrames={30 * 90}  // 默认 90 秒，运行时覆盖
    fps={30}

主合成组件 (src/VideoComposition.tsx)

import { AbsoluteFill, Audio, Sequence, useVideoConfig } from 'remotion';
import { TransitionSeries, linearTiming } from '@remotion/transitions';
import { fade, slide, wipe, zoom } from '@remotion/transitions/presentations';
import { Scene } from './components/Scene';
import { AnimatedSubtitle } from './components/Subtitle';
import { getTemplate } from './templates';

const transitionMap = {
  fade: fade(), slide_left: slide({ direction: 'from-right' }),
  slide_right: slide({ direction: 'from-left' }),
  slide_up: slide({ direction: 'from-bottom' }),
  zoom_in: zoom(), zoom_out: zoom(), wipe: wipe(), none: fade(),
};

export const VideoComposition = ({ script, audioResults }) => {
  const { fps } = useVideoConfig();
  const template = getTemplate(script.template);

  let currentFrame = 0;

  return (
    <AbsoluteFill style={{ backgroundColor: template.bg_color }}>
      <TransitionSeries>
        {script.scenes.map((scene, i) => {
          const audio = audioResults[scene.scene_id];
          const durationInFrames = Math.ceil(audio.audio_duration * fps);
          const sceneStart = currentFrame;
          currentFrame += durationInFrames;

          return (
            <React.Fragment key={scene.scene_id}>
              <TransitionSeries.Sequence durationInFrames={durationInFrames}>
                <AbsoluteFill>
                  <Scene scene={scene} template={template} />
                  <AnimatedSubtitle
                    words={audio.word_boundaries}
                    sceneStartFrame={sceneStart}
                    style={script.subtitle_style}
                  />
                </AbsoluteFill>
                <Audio src={audio.audio_file} />
              </TransitionSeries.Sequence>
              {i < script.scenes.length - 1 && (
                <TransitionSeries.Transition
                  presentation={transitionMap[scene.transition_out]}
                  timing={linearTiming({ durationInFrames: Math.floor(fps * 0.5) })}
                />
              )}
            </React.Fragment>
          );
        })}
      </TransitionSeries>

      // 全局背景音乐
      {script.bgm && (
        <Audio src={script.bgm.file} volume={script.bgm.volume} />
      )}
    </AbsoluteFill>
  );
};

渲染 API (scripts/render.ts)

import { bundle } from '@remotion/bundler';
import { renderMedia, selectComposition } from '@remotion/renderer';
import path from 'path';

export async function renderVideo(
  script: ScriptData,
  audioResults: Map<number, TTSResult>,
  outputPath: string
): Promise<void> {
  // Step 1: 打包 Remotion 项目
  const bundleLocation = await bundle({
    entryPoint: path.resolve(__dirname, '../src/index.ts'),
  });

  // Step 2: 计算总帧数
  const totalFrames = script.scenes.reduce((sum, s) => {
    const audio = audioResults.get(s.scene_id)!;
    return sum + Math.ceil(audio.audio_duration * script.fps);
  }, 0);

  // Step 3: 获取 Composition
  const composition = await selectComposition({
    serveUrl: bundleLocation,
    id: 'MainVideo',
    inputProps: { script, audioResults: Object.fromEntries(audioResults) },
  });

  // Step 4: 渲染输出
  await renderMedia({
    serveUrl: bundleLocation,
    composition: {
      ...composition,
      durationInFrames: totalFrames,
      width: script.resolution.width,
      height: script.resolution.height,
      fps: script.fps,
    },
    codec: 'h264',
    outputLocation: outputPath,
    inputProps: { script, audioResults: Object.fromEntries(audioResults) },
    onProgress: ({ progress }) => {
      process.stdout.write(`\r渲染进度: ${(progress * 100).toFixed(1)}%`);
    },
  });

  console.log(`\n✅ 渲染完成: ${outputPath}`);
}

ChartScene 实现要点

// ECharts 在 Remotion 中必须使用 SVG 渲染模式（同步，确定性）
// 不能用 canvas 模式（异步 requestAnimationFrame 会导致帧不同步）

import { useEffect, useRef, useState } from 'react';
import { useCurrentFrame, useVideoConfig, interpolate, delayRender, continueRender } from 'remotion';
import * as echarts from 'echarts';

export const ChartScene = ({ scene, template }) => {
  const containerRef = useRef<HTMLDivElement>(null);
  const chartRef = useRef<echarts.ECharts>();
  const [handle] = useState(() => delayRender('Loading chart'));
  const frame = useCurrentFrame();
  const { fps } = useVideoConfig();

  useEffect(() => {
    if (!containerRef.current) return;
    chartRef.current = echarts.init(containerRef.current, undefined, {
      renderer: 'svg',  // SVG 模式：同步、确定性
    });

    const option = buildChartOption(scene.visual.data, template, frame, fps);
    chartRef.current.setOption(option);

    const timer = setTimeout(() => continueRender(handle), 100);
    return () => { clearTimeout(timer); chartRef.current?.dispose(); };
  }, [frame]);

  return <div ref={containerRef} style={{ width: '100%', height: '100%' }} />;
};

function buildChartOption(data, template, frame, fps) {
  const progress = interpolate(frame, [0, fps], [0, 1], {
    extrapolateRight: 'clamp',
  });

  return {
    ...data,
    animation: false,
    color: template.chart_colors,
    // 用 progress 驱动数据展示进度
    series: data.series?.map(s => ({
      ...s,
      data: s.data.map(v =>
        typeof v === 'number' ? v * progress : v
      ),
    })),
  };
}
关键注意：ECharts 在 Remotion 中必须使用 SVG 渲染模式。Canvas 模式使用 requestAnimationFrame 异步渲染，会导致帧捕获时画面不完整。SVG 模式是同步的，确保每帧截图时图表已完整渲染。

7.4 字幕动画组件

字幕直接作为 React 组件渲染在画面上，利用 TTS 返回的词级时间戳实现逐字高亮和动画效果。

核心实现 (src/components/Subtitle.tsx)

import {
  useCurrentFrame, useVideoConfig, interpolate, spring, Easing
} from 'remotion';

interface SubtitleProps {
  words: WordBoundary[];
  sceneStartFrame: number;
  style: SubtitleStyle;
}

export const AnimatedSubtitle = ({ words, sceneStartFrame, style }) => {
  const frame = useCurrentFrame();
  const { fps } = useVideoConfig();

  const relativeFrame = frame - sceneStartFrame;
  const currentTimeMs = relativeFrame / fps * 1000;

  const positionStyle = {
    position: 'absolute' as const,
    bottom: style.position === 'bottom' ? style.offset_y : undefined,
    top: style.position === 'top' ? style.offset_y : undefined,
    left: 0, right: 0,
    textAlign: 'center',
    maxWidth: style.max_width,
    margin: '0 auto',
    fontFamily: style.font_family,
    fontSize: style.font_size,
    fontWeight: 700,
    lineHeight: 1.4,
    textShadow: `-2px -2px 0 ${style.stroke_color}, `
      + `2px -2px 0 ${style.stroke_color}, `
      + `-2px 2px 0 ${style.stroke_color}, `
      + `2px 2px 0 ${style.stroke_color}`,
  };

  return (
    <div style={positionStyle}>
      {words.map((word, i) => {
        const wordStartMs = word.offset_ms;
        const wordEndMs = word.offset_ms + word.duration_ms;
        const isActive = currentTimeMs >= wordStartMs && currentTimeMs < wordEndMs;
        const isPast = currentTimeMs >= wordEndMs;

        // 计算动画值
        const wordStartFrame = sceneStartFrame + Math.floor(wordStartMs / 1000 * fps);
        let scale = 1;
        let translateY = 0;

        if (isActive) {
          switch (style.animation) {
            case 'bounce':
              scale = spring({
                frame: frame - wordStartFrame, fps,
                config: { damping: 12, stiffness: 200, mass: 0.8 },
              });
              break;
            case 'pop':
              scale = interpolate(frame - wordStartFrame, [0, 3, 6], [0.5, 1.2, 1.0], {
                extrapolateRight: 'clamp',
              });
              break;
            case 'slide':
              translateY = interpolate(frame - wordStartFrame, [0, 5], [20, 0], {
                extrapolateRight: 'clamp', easing: Easing.out(Easing.cubic),
              });
              break;
            case 'fade':
              scale = interpolate(frame - wordStartFrame, [0, 4], [0.8, 1], {
                extrapolateRight: 'clamp',
              });
              break;
          }
        }

        const color = isActive ? style.highlight_color : style.color;

        return (
          <span key={i} style={{
            display: 'inline-block',
            color,
            transform: `scale(${scale}) translateY(${translateY}px)`,
            margin: '0 1px',
          }}>
            {word.text}
          </span>
        );
      })}
    </div>
  );
};

动画效果说明

动画效果实现
bounce当前词弹跳放大spring() 物理弹性
pop快速弹出再收回interpolate 三段插值
slide从下方滑入translateY + Easing
fade轻微缩放淡入interpolate scale

变色高亮：当前播放到的词用 highlight_color，已读和未读词用 color，实现逐字高亮跟随效果。

7.5 转场效果系统

使用 @remotion/transitions 的 TransitionSeries 实现场景间过渡，支持 8 种转场类型：

转场类型效果实现 API
fade淡入淡出fade()
slide_left从右向左滑入slide({ direction: 'from-right' })
slide_right从左向右滑入slide({ direction: 'from-left' })
slide_up从下向上滑入slide({ direction: 'from-bottom' })
zoom_in缩放放大过渡zoom()
zoom_out缩放缩小过渡zoom()
wipe擦除过渡wipe()
bounce弹性弹入自定义 spring 动画
none无转场（硬切）—

转场时长

默认转场持续 0.5 秒（15帧 @ 30fps），可通过 linearTiming({ durationInFrames: N }) 调整。转场帧数从相邻两个场景中各取一半重叠渲染。

素材管理器 (scripts/assets.ts)

import fs from 'fs/promises';
import path from 'path';
import crypto from 'crypto';

const CACHE_DIR = path.resolve(__dirname, '../output/assets');

export class AssetManager {
  constructor(private pexelsKey: string, private unsplashKey: string) {}

  async fetchImage(query: string, source: 'pexels' | 'unsplash' = 'pexels'): Promise<string> {
    const cacheKey = crypto.createHash('md5').update(`${source}_${query}`).digest('hex');
    const cachedPath = path.join(CACHE_DIR, `${cacheKey}.jpg`);

    // 检查缓存
    try {
      await fs.access(cachedPath);
      return cachedPath;
    } catch { // cache miss }

    // 获取图片 URL
    const imageUrl = source === 'pexels'
      ? await this.searchPexels(query)
      : await this.searchUnsplash(query);

    // 下载并缓存
    const buffer = await fetch(imageUrl).then(r => r.arrayBuffer());
    await fs.mkdir(path.dirname(cachedPath), { recursive: true });
    await fs.writeFile(cachedPath, Buffer.from(buffer));

    return cachedPath;
  }

  private async searchPexels(query: string): Promise<string> {
    const res = await fetch(`https://api.pexels.com/v1/search?query=${encodeURIComponent(query)}&per_page=1`, {
      headers: { Authorization: this.pexelsKey },
    });
    const data = await res.json();
    return data.photos[0]?.src?.large2x || data.photos[0]?.src?.large;
  }

  async prepareAssets(script: ScriptData): Promise<void> {
    for (const scene of script.scenes) {
      const v = scene.visual;
      if (v.type === 'text_card' && v.bg_image?.startsWith('pexels:')) {
        const query = v.bg_image.replace('pexels:', '');
        v.bg_image = await this.fetchImage(query, 'pexels');
      }
      if (v.type === 'image' && v.image_source?.startsWith('pexels:')) {
        const query = v.image_source.replace('pexels:', '');
        v.image_source = await this.fetchImage(query, 'pexels');
      }
    }
  }
}

素材标识规则：在 script.json 中用前缀指定素材源，如 "pexels:sky_blue_clouds" 表示从 Pexels 搜索关键词 sky_blue_clouds 的图片。素材管理器解析前缀后自动获取并缓存。

7.8 流水线编排模块

编排器是整个流水线的入口，串联所有模块，从 script.json 到最终 MP4：

编排器 (scripts/pipeline.ts)

import { generateAllAudio } from './tts';
import { AssetManager } from './assets';
import { renderVideo } from './render';
import { validateScript } from './validate';
import path from 'path';

export async function generateVideo(scriptPath: string): Promise<string> {
  const outputDir = path.resolve('./output');
  const audioDir = path.join(outputDir, 'audio');
  const finalDir = path.join(outputDir, 'final');

  // Step 0: 加载并校验脚本
  console.log('📋 Step 0: 加载并校验 script.json...');
  const script = await loadScript(scriptPath);
  validateScript(script);

  // Step 1: TTS 语音合成（并行）
  console.log('🎙️  Step 1: 生成 TTS 语音...');
  const audioResults = await generateAllAudio(script, audioDir);

  // Step 2: 更新场景时长
  console.log('⏱️  Step 2: 计算实际时长...');
  const scriptWithTiming = updateDurations(script, audioResults);

  // Step 3: 素材准备
  console.log('🖼️  Step 3: 准备素材...');
  const assetManager = new AssetManager(process.env.PEXELS_KEY!, process.env.UNSPLASH_KEY!);
  await assetManager.prepareAssets(scriptWithTiming);

  // Step 4: Remotion 渲染
  console.log('🎬 Step 4: Remotion 渲染中...');
  const outputPath = path.join(finalDir, `${script.video_id}.mp4`);
  await renderVideo(scriptWithTiming, audioResults, outputPath);

  console.log(`\n✅ 视频生成完成: ${outputPath}`);
  return outputPath;
}

// CLI 入口
const scriptPath = process.argv[2];
if (scriptPath) {
  generateVideo(scriptPath).catch(console.error);
}
